\begin{frame}{Pseudocódigo}
\begin{algorithm}[H]
\caption{Classificador Baseado em Clustering}
\begin{algorithmic}[1]
\REQUIRE $X_{treino}$, $y_{treino}$, $X_{teste}$
\STATE Normalizar dados com StandardScaler
\STATE Encontrar $k_{ótimo}$ usando método do cotovelo
\STATE Aplicar clustering com $k_{ótimo}$
\STATE Mapear clusters $\rightarrow$ classes (mais frequente)
\STATE Classificar novos pontos usando mapeamento
\RETURN Predições
\end{algorithmic}
\end{algorithm}
\end{frame}

\begin{frame}{Método do Cotovelo}
\begin{columns}
\begin{column}{0.6\textwidth}
\begin{figure}
\centering
\includegraphics[width=\textwidth]{../img/elbow_method_example.png}
\caption{Exemplo do método do cotovelo}
\end{figure}
\end{column}
\begin{column}{0.4\textwidth}
\textbf{Características:}
\begin{itemize}
    \item Determina $k$ automaticamente
    \item Baseado na inércia
    \item Identifica "cotovelo" na curva
    \item Evita overfitting
\end{itemize}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Protocolo Experimental}
\begin{block}{Configuração dos Experimentos}
\begin{itemize}
    \item \textbf{Repetições}: 30 execuções (sementes 1-30)
    \item \textbf{Divisão}: 70\% treino / 30\% teste
    \item \textbf{Normalização}: StandardScaler
    \item \textbf{Estratificação}: Manter proporção das classes
\end{itemize}
\end{block}

\vspace{0.5cm}

\begin{block}{Métricas de Avaliação}
\begin{itemize}
    \item Acurácia média e desvio padrão
    \item Matriz de confusão
    \item Tempo de execução
    \item Número de clusters utilizados
\end{itemize}
\end{block}
\end{frame}
